			CREATE PROCEDURE [dbo].[usp_GetMRDFReceivedBatches]
			AS
			BEGIN

			SELECT B.PrintBatch,DT.DocTypeCode,C.AppCode,BU.CustCode,CONVERT(VARCHAR(10),B.PrintBatch) as BatchCode 
			FROM [PnS].BatchSummary B
			INNER JOIN [AM].DocTypes DT ON DT.Id=B.DocTypeId 
			INNER JOIN [Common].Clients C ON C.Id=B.ClientId
			INNER JOIN [Common].BusinessUnit BU ON BU.Id=C.BusinessUnit
			WHERE B.IsMRDFReceived=1 AND B.IsPDFsReceived=0
			END

				ELSE If(@mailPieceStatus='INTERNAL')--This will be used for getting only submitted records to PCI
				BEGIN
					SELECT JobId, PieceId, [Date], TotalSheets, InputWeight, Feeder3_Sel, Feeder4_Sel
					, Feeder5_Sel, LowCap_Sel, NoSeal_Sel, Feeder2_Sel, ZipMark_Sel, Address1, Address2

			CREATE PROCEDURE [dbo].[usp_GetPullBackRequestDataByJobId]
			@jobId int
			AS
			BEGIN
				-- SET NOCOUNT ON added to prevent extra result sets from
				-- interfering with SELECT statements.
				SET NOCOUNT ON;
				
				BEGIN TRAN
				--Pulled back logic
				UPDATE [PnS].[LogBatchTimeLine]
				SET
				[PCIBatchPrepEnd]=CAST('1753-1-1' AS DATETIME)
				WHERE BatchId=@jobId

				DECLARE @PnSBatchUniqueId UNIQUEIDENTIFIER,@AMPrintBatchId INT

				SET @PnSBatchUniqueId=(SELECT TOP 1 BatchIdentifier FROM PnS.BatchSummary WHERE PrintBatch=@jobId)

				SET @AMPrintBatchId=(SELECT TOP 1 Id FROM AM.PrintBatches WHERE UniqueID=@PnSBatchUniqueId)

				SELECT * FROM [AM].[PullBackRequests] 
				WHERE MailPieceUniqueId IN (SELECT UniqueId FROM AM.MailPieces WHERE PrintBatchId=@AMPrintBatchId)

				COMMIT TRAN

			END

				-- Description:	 Get mail pieces list which are not yet picked up
				-- =============================================
				CREATE PROCEDURE [dbo].[uspweb_FedExPickedUpReport]
					@deploymentDate datetime,
					@lastReceivedOnDate datetime,
					@shippingType varchar(128),
					@isPickedUp int--Possible values : 1/0/-1'
				AS
					if(@isPickedUp=0)
					BEGIN
						SELECT * FROM AM.MailPieces MP
						INNER JOIN AM.MailBatches MB ON MB.ID=MP.MailBatchId
						INNER JOIN AM.DocTypes D ON D.Id=MP.DocTypeId
						WHERE MB.ReceivedOn >= @deploymentDate AND MB.ReceivedOn < @lastReceivedOnDate
						AND MP.ShippingStatus <> 'Error' AND MP.IsPickedUp=0
						AND D.ShippingCarrier=@shippingType
					END
					ELSE IF(@isPickedUp=1)
					BEGIN
						SELECT * FROM AM.MailPieces MP
						INNER JOIN AM.MailBatches MB ON MB.ID=MP.MailBatchId
						INNER JOIN AM.DocTypes D ON D.Id=MP.DocTypeId
						WHERE MB.ReceivedOn >= @deploymentDate AND MB.ReceivedOn < @lastReceivedOnDate
						AND MP.ShippingStatus <> 'Error' AND MP.IsPickedUp=1
						AND D.ShippingCarrier=@shippingType
					END
					ELSE
					BEGIN
						SELECT * FROM AM.MailPieces MP
						INNER JOIN AM.MailBatches MB ON MB.ID=MP.MailBatchId
						INNER JOIN AM.DocTypes D ON D.Id=MP.DocTypeId
						WHERE MB.ReceivedOn >= @deploymentDate AND MB.ReceivedOn < @lastReceivedOnDate
						AND MP.ShippingStatus <> 'Error'
						AND D.ShippingCarrier=@shippingType
					END
				GO

					select	ISNULL(sc.SweepName,pc.SweepName) as BatchName,
							ISNULL(pc.PendingCount,0) as PendingCount,
							ISNULL(sc.ScannedCount,0) as ScannedCount
					from sc full outer join pc on sc.SweepName=pc.SweepName
				)
				
				select	ISNULL(t.BatchName,sw.BatchName) as BatchName,
						IsNULL(t.ScannedCount,0) as ScannedLoans,
						(ISNULL(t.PendingCount,0) + ISNULL(t.ScannedCount,0)) as TotalLoans 
						from t
						full outer join sw on t.BatchName = sw.BatchName
			END

			ALTER PROCEDURE [dbo].[SP_RegisterScan]
			-- Add the parameters for the stored procedure here
			@trackno varchar(255),
			@clientCode varchar(128),
			@operatorid varchar(50),
			@machineid varchar(255),
			@sweepName varchar(255)-- 05PM, 06PM, 07PM, Backlog
			--@todayESTDate date
			AS
			BEGIN
			declare	@jobid int,
				@pieceid int,
				@mrdfreqdate date,
				@ispullbackrequested bit,
				@isqcrequested bit,
				@disposition varchar(50),
				@mailpiecestatus varchar(255),
				@totalsheet int = null,
				@pdffilename varchar(255),
				@printlocationcl varchar(1024),
				@inprogresscnt int,
				@mailpieceid int,
				@code varchar(50),
				@existjobid int,
				@jobName varchar(512),
				@serviceType varchar(255),
				@isMPSUpdate bit,
				@clientAppCode varchar(128),
				@clientName varchar(255),
				@swp varchar(255)

			BEGIN TRY

				set @trackno = replace(@trackno, 'AUT','')
				set @clientCode = replace(@clientCode, 'AUT','')
				--Get Print Location name
				SET @printlocationcl=(select Value from [Common].[SystemSettings] where Id ='Common_PrintLocationCL')
				-- Get mailpiece, MRDFRequest data filter by trackingno & printlocation

				if(@sweepName is null)
				begin 
					select 'Invalid' as Code, NULL as TrackingNo, 'Batch name can not be null' as [Message], NULL as [JobName]
					return
				end
				select @mailpieceid= Id,@pieceid=PieceId,@jobid=JobId,@mrdfreqdate=MRDFReqDate,@ispullbackrequested =IsPullBackRequested,@isqcrequested=IsQCRequested,@totalsheet=TotalSheets,@pdffilename=PDFFileName,@jobName=JobName,@serviceType=ShippingServiceType,@clientAppCode=AppCode,@clientName=ClientName,@swp=SweepName
				from dbo.vw_DallasBatches		
				where TrackingNo=@trackno and PrintLocation=@printlocationcl --and b.IsSubmitted=1

				if(@mailpieceid IS NULL) 
				begin
					select 'Invalid' as Code, NULL as TrackingNo, NULL as [Message], NULL as [JobName]
					return
				end

				--Validate tracking no is assign to this client
				if(@clientCode <> @clientAppCode)
				begin
					select 'Invalid' as Code, @trackno as TrackingNo, REPLACE('This tracking number belongs to @clientName client. Please check your document.','@clientName',@clientName) as [Message], NULL as [JobName]
					return
				end
				-- Validate this mailpiece is assign to this batch or not.
				if(@swp<>@sweepName)
				begin
					select 'Invalid' as Code, @trackno as TrackingNo, REPLACE('This document belongs to <span style="color:#000"> @swp </span> batch.','@swp',@swp) as [Message], NULL as [JobName]
					return
				end
				
				if exists(select 1 from PnS.MRDFResponse where JobId = @jobid and PieceId = @pieceid) 
				begin
					select 'Rescan' as Code, @trackno as TrackingNo, NULL as [Message], @jobName as [JobName]
					return
				end
				-- set mailpiece status if ispullbackrequested then update otherwise not
				SET @mailpiecestatus='Shipped'

				if(@ispullbackrequested = 1) 
				begin
					SET @code = 'Pulledback'
					SET @disposition='4' 
					SET @mailpiecestatus='PulledBack'
				end
				else if(@isqcrequested=1)
				begin
					SET @disposition='8'
					SET @code = 'QC'
				end
				else
				begin
					SET @disposition='5'
					SET @code = 'Valid'
				end

				-- Validate if trackno is same as MailPiece and service type is same as USPS_EXPRESSMAIL. 
				-- Then do not update MailPieceStatus
				set @isMPSUpdate=1;
				if(@trackno = cast(@mailpieceid as varchar) and @serviceType = 'USPS_EXPRESSMAIL') set @isMPSUpdate=0;
				
				-- Start Transaction
				BEGIN TRANSACTION	
					-- insert data into MRDFResponse table
					INSERT INTO [PnS].[MRDFResponse] ([JobId],[PieceId],[Date],[Feeder2_Fed],[OperatorId],[MachineId],									[ProcessingTimeStamp],[DetailedErrorId],[Disposition],[Input_Fed],[Feeder3_Fed],[Feeder4_Fed],[Feeder5_Fed],						[ComputedWeight],[MeterDate],[StatusSource],[PostageMetered]
					,[WeightMeasured],[ExitLocation],[TotalSheets],[PDFFileName],[PackageTrackingNumber])
					VALUES(@jobid,@pieceid,@mrdfreqdate,0,@operatorid,@machineid,GETUTCDATE(),'4691'
					,@disposition,'00','0','0','0','0.000','00000000', @printlocationcl,'0.000','0.000',@printlocationcl
					,@totalsheet,@pdffilename,NULL)
						
					-- update mailepiecestatus depend on [Disposition] value
					if(@isMPSUpdate = 1)
					begin
						Update [PnS].[MRDFRequest] SET MailPieceStatus = @mailpiecestatus
						where JobId=@jobid and PieceId = @pieceid

						-- Check if we are yet to received response for any mail pieces
						select @inprogresscnt= count(*) from [PnS].[MRDFRequest] where MailPieceStatus='InProgress' and JobId=@jobid
					
						-- If responses are received for entire batch, then update batch status to received
						if(@inprogresscnt = 0)
						begin
							update PnS.BatchSummary set PipelineStatus = 'MRDFReceived', MRDFReceivedOn = GETUTCDATE() where PrintBatch = @jobid
						end
					end
				--End Transaction
				COMMIT
				
				select @code as Code, @trackno as TrackingNo, NULL as [Message], @jobName as [JobName]
			END TRY
			BEGIN CATCH
				IF @@TRANCOUNT > 0
					ROLLBACK
			END CATCH
			END

	ALTER PROCEDURE [dbo].[usp_GetNextBatchIdByJobName] 
		-- Add the parameters for the stored procedure here
		@jobName varchar(50),
		@operation varchar(50)
	AS
	BEGIN
		
		IF (@jobName = 'PnSJobMonitor' AND @operation = 'ReportBack')
		BEGIN
			SELECT PrintBatch AS BatchId FROM [PnS].BatchSummary WHERE PipelineStatus = 'ReportPrepped' AND  IsReportPrepped = 1 AND IsReportedBack = 0
		END
		ELSE IF(@jobName = 'PnSCoreService')
		BEGIN
			IF(@operation = 'Prep')
			BEGIN
				SELECT PrintBatch AS BatchId FROM PnS.BatchSummary WHERE PipelineStatus = 'Downloaded' AND  IsDownloaded = 1 AND IsPrepped = 0 
			END
			ELSE IF(@operation = 'ProcessMRDF')
			BEGIN
				SELECT PrintBatch AS BatchId FROM PnS.BatchSummary WHERE PipelineStatus = 'MRDFReceived' AND  IsMRDFReceived = 1 AND IsReportPrepped = 0
			END
		END
		ELSE IF(@jobName = 'PCIJobMonitor' AND @operation = 'Submit')
		BEGIN
			SELECT PrintBatch AS BatchId FROM PnS.BatchSummary WHERE PipelineStatus = 'Prepped' AND  IsPrepped = 1 AND IsSubmitted = 0 
		END
	END

	(
	@EstDate DATE
	)
	RETURNS TABLE 
	AS
	RETURN
	(
	with db as
	(
		select * from [dbo].[ufn_GetDallasBatches](@EstDate)
	),
	q as
	(
	SELECT m.Id,m.BusinessFriendlyIdValue as LoanNumber,m.ClientId, m.PrintLocation, m.EffectiveReceivedOnEst, m.TrackingNo,m.IsPullBackRequested,m.IsQCRequested,m.TotalSheets , m.DueDate,m.TotalPages,m.ShippingStatus,

	d.ShippingServiceType,d.ShippingCarrier,d.IsBRE,
	c.AppCode,c.Name as ClientName,(select BatchName from db where EffectiveReceivedOnEst between StartDate AND EndDate AND ShippingCarrier=d.ShippingCarrier AND ShippingServiceType=d.ShippingServiceType AND AppCode=c.AppCode) BatchName
	FROM [AM].[MailPieces] m
	LEFT OUTER JOIN PnS.MRDFRequest req ON m.UniqueId = req.UniqueId
	LEFT OUTER JOIN PnS.BatchSummary b ON req.JobId = b.PrintBatch
	LEFT OUTER JOIN PnS.MRDFResponse res ON req.JobId = res.JobId AND req.PieceId=res.PieceId
	INNER JOIN AM.DocTypes d ON m.DocTypeId=d.Id
	INNER JOIN Common.Clients c ON m.ClientId=c.Id
	WHERE m.PrintLocation=(SELECT Value FROM Common.SystemSettings WHERE Id='Common_PrintLocationCL') 
	AND m.EffectiveReceivedOnEst >=(select MIN(StartDate) from db)
	)
	select * from q where BatchName is not null 
	)

		FROM [PnS].[MRDFRequest] Req
		INNER JOIN PnS.BatchSummary BS ON REQ.JobId=BS.PrintBatch
		LEFT OUTER JOIN AM.MailPieces MP ON REQ.UniqueId=MP.UniqueId
		WHERE Req.PrintTrailBatchId = @printTrailBatchId
	END
	ELSE If(@mailPieceStatus='INTERNAL')--This will be used for getting only submitted records to PCI
	BEGIN
		SELECT JobId, PieceId, [Date], TotalSheets, InputWeight, Feeder3_Sel, Feeder4_Sel
		, Feeder5_Sel, LowCap_Sel, NoSeal_Sel, Feeder2_Sel, ZipMark_Sel, Address1, Address2
		, Zip5, Zip4, Zip2, DivertVS2, DivertVS3
		, DivertVS4, ReturnLogoBitMap, ReturnLine1, ReturnLine2, ReturnLine3, ReturnLine4
		, ReturnLine5, ReturnLine6, GroupNumber, SocialSecNumber, CheckAmount, PassThru
		, Expansion, EndofRecord, UniqueId, PDFFileName,ErrId, ErrDesc, MailPieceStatus, IsPDFReceived,PrintTrailBatchId 
		FROM [PnS].[MRDFRequest] WHERE PrintTrailBatchId = @printTrailBatchId AND ErrId NOT IN (-1,-2,-3)
	END
	ELSE
	BEGIN
		SELECT REQ.*,MP.Id as MailPieceId,MP.EffectiveReceivedOnEst, BS.ClientId,BS.DocTypeId,D.ShippingCarrier,
		D.ShippingServiceType,C.AppCode,MP.DueDate as MailPieceDueDate,D.EnvelopeId,D.BREEnvelopeId
		FROM PnS.MRDFRequest REQ
		INNER JOIN PnS.BatchSummary BS ON REQ.JobId=BS.PrintBatch
		LEFT OUTER JOIN AM.MailPieces MP ON REQ.UniqueId=MP.UniqueId
		INNER JOIN AM.DocTypes D ON BS.DocTypeId=D.Id
		INNER JOIN Common.Clients C ON C.Id=BS.ClientId
		WHERE REQ.PrintTrailBatchId=@printTrailBatchId AND ISNULL(MailPieceStatus,'')=@mailPieceStatus 
	END
END

ALTER PROCEDURE [dbo].[usp_GetMRDFRequestDataByPrintTrailId] 
@printTrailId INT
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

 SELECT REQ.*,MP.Id as MailPieceId,MP.EffectiveReceivedOnEst, BS.ClientId,BS.DocTypeId,D.ShippingCarrier,
 D.ShippingServiceType,C.AppCode,MP.DueDate as MailPieceDueDate,D.EnvelopeId,D.BREEnvelopeId
 FROM PnS.MRDFRequest REQ
 INNER JOIN PnS.BatchSummary BS ON REQ.JobId=BS.PrintBatch
 LEFT OUTER JOIN AM.MailPieces MP ON REQ.UniqueId=MP.UniqueId
 INNER JOIN AM.DocTypes D ON BS.DocTypeId=D.Id
 INNER JOIN Common.Clients C ON C.Id=BS.ClientId
 WHERE BS.PipelineStatus='Prepped' AND  IsPrepped = 1 AND IsSubmitted = 0
 AND BS.PrintTrailId=@printTrailId
	
END
