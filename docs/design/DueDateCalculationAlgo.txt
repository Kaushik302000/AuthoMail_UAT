private DateTime GetDueDate(int slaDays, TimeSpan estSLACutOffTime, DateTime utcReceivedOn, string shippingCarrier)
{

	//ReceivedOn Time of Batch
	var estReceivedOn = utcReceivedOn.UtcToEst();
	estReceivedOn.Dump();
	"==ReceivedOnDate EST==".Dump();
	var estReceivedOnDate = estReceivedOn.Date;
	
	//logic for FedEx Saturday Cut off timings
	if (estReceivedOnDate.DayOfWeek.ToString() == "Saturday" && shippingCarrier == "FedEx")
	{
		estSLACutOffTime = new TimeSpan(12,00,00);
	}

	//Should be in cut off Date time
	var estSLACutOffDateTime = (estReceivedOnDate + estSLACutOffTime);
	
	estSLACutOffDateTime.Dump();
	"==Batch CutOffTime EST==".Dump();
	
	if (estReceivedOn > estSLACutOffDateTime)
	{
		//Here we have actual SLA days for DocType and added 1 because batch received after cut off time.
		slaDays += 1;
	}
	
	//Sample Holiday List
	List<DateTime> holidayList=new List<DateTime>();
	holidayList.Add(Convert.ToDateTime("6/25/2018 00:00:00 AM"));
	//holidayList.Add(Convert.ToDateTime("6/26/2018 00:00:00 AM"));
	holidayList.Add(Convert.ToDateTime("6/27/2018 00:00:00 AM"));
	holidayList.Add(Convert.ToDateTime("6/29/2018 00:00:00 AM"));
	
	//Here we just add actual SLA days because batch received before cut off time
	var dueDate = estReceivedOnDate.AddDays(slaDays);
	dueDate.Dump();
	"==Actual Due Date EST==".Dump();
	bool isHoliday=true;
	while (isHoliday)
	{
		if(holidayList.Contains(dueDate))
		{
			dueDate=dueDate.AddDays(1);
		}
		else if(dueDate.DayOfWeek.ToString()=="Sunday")
		{
			dueDate=dueDate.AddDays(1);
		}
		else
			isHoliday=false;
	}
	"Final DueDate EST".Dump();
	(dueDate + new TimeSpan(23, 59, 59)).Dump();
	return (dueDate + new TimeSpan(23, 59, 59)).EstToUtc();
}

public static class DateExtentions
{
		public static DateTime AnyTimeZoneConvert(this DateTime dt, string fromTimeZone, string toTimeZone)
        {
            dt = DateTime.SpecifyKind(dt, DateTimeKind.Unspecified);
            return TimeZoneInfo.ConvertTime(dt, TimeZoneInfo.FindSystemTimeZoneById(fromTimeZone), TimeZoneInfo.FindSystemTimeZoneById(toTimeZone));
        }
		
		public static DateTime UtcToClientTimeZone(this DateTime dt, string clientTimeZone)
        {
            dt = DateTime.SpecifyKind(dt, DateTimeKind.Unspecified);
            return TimeZoneInfo.ConvertTime(dt, TimeZoneInfo.FindSystemTimeZoneById("UTC"), TimeZoneInfo.FindSystemTimeZoneById(clientTimeZone));
        }

        public static DateTime ClientTimeZoneToUtc(this DateTime dt, string clientTimeZone)
        {
            dt = DateTime.SpecifyKind(dt, DateTimeKind.Unspecified);
            return TimeZoneInfo.ConvertTime(dt, TimeZoneInfo.FindSystemTimeZoneById(clientTimeZone), TimeZoneInfo.FindSystemTimeZoneById("UTC"));
        }

        public static DateTime UtcToEst(this DateTime dt)
        {
            return dt.UtcToClientTimeZone("Eastern Standard Time");
        }

        public static DateTime EstToUtc(this DateTime dt)
        {
            return dt.ClientTimeZoneToUtc("Eastern Standard Time");
        }
}
